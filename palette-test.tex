      at (9mm, 11mm) {#2};
    \node[anchor=center, text=white, font=\fontsize{6}{7}\selectfont]
      at (9mm, 3.5mm) {#3};
  \end{tikzpicture}%
}

{\Large\bfseries HSB Palette (true yellow for Air)}

\vspace{4mm}

% HSB palette: anchor element colors at cardinal signs, interpolate between
% Aries (Fire) = red, H=0
% Cancer (Water) = blue, H=240
% Libra (Air) = yellow, H=60
% Capricorn (Earth) = dark green, H=120, B=0.5
%
% Aries→Taurus→Gemini→Cancer: 0° → 240° going through 300° (magenta/violet)
% Cancer→Leo→Virgo→Libra: 240° → 60° going through 150° (teal/green)
% Libra→Scorpio→Sagittarius→Capricorn: 60° → 120° (yellow-green)
% Capricorn→Aquarius→Pisces→Aries: 120° → 0°/360° going through 270° (dark tones)

\definecolor{hsb-aries}{Hsb}{0, 1, 1}         % red (Fire)
\definecolor{hsb-taurus}{Hsb}{320, 1, 0.85}    % magenta-rose
\definecolor{hsb-gemini}{Hsb}{280, 0.8, 0.8}   % violet
\definecolor{hsb-cancer}{Hsb}{240, 1, 1}        % blue (Water)
\definecolor{hsb-leo}{Hsb}{190, 0.9, 0.85}      % teal-cyan
\definecolor{hsb-virgo}{Hsb}{140, 0.8, 0.7}    % green
\definecolor{hsb-libra}{Hsb}{60, 1, 1}          % yellow (Air)
\definecolor{hsb-scorpio}{Hsb}{80, 0.9, 0.7}   % yellow-green
\definecolor{hsb-sagittarius}{Hsb}{100, 0.9, 0.6} % green
\definecolor{hsb-capricorn}{Hsb}{120, 1, 0.5}   % dark green (Earth)
\definecolor{hsb-aquarius}{Hsb}{270, 0.7, 0.45}  % dark violet
\definecolor{hsb-pisces}{Hsb}{340, 0.8, 0.55}    % dark rose
